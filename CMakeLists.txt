cmake_minimum_required(VERSION 3.13)

include (pico_sdk_import.cmake)

project(test_project C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()

# The original build enabled no warnings at all, which is how a printf with a
# %s and no argument survived in the EEPROM error path. Must precede
# add_executable() to apply to the target.
add_compile_options(-Wall -Wextra -Wno-unused-parameter)

# Commit id for the firmware version string. Derived at BUILD time, not
# configure time. This used to run during configure and cache nothing, which
# looked fine but wasn't: commit, then `make` without re-running `cmake`, and
# the id kept the previous configure's value. The tree was genuinely clean at
# that point, so no trailing '+' warned of the mismatch, and a deployed board
# pointed at the wrong commit -- the exact inverse of what the id is for. A
# custom target now re-runs the derivation on every build via `cmake -P`, so
# the value is recomputed whether or not CMake reconfigured.
#
# Preferred over a CMAKE_CONFIGURE_DEPENDS hook on .git/HEAD + .git/index for
# two reasons: (1) a new untracked file changes neither path, so that hook
# would miss the '+' the checkout earns for it; (2) in a git worktree -- which
# this repo uses -- .git is a *file* and HEAD lives under .git/worktrees/..,
# so the naive paths do not even exist. Recomputing unconditionally sidesteps
# both.
#
# In a checkout the id is read straight from git, so it is self-verifying. The
# Docker build has no .git and passes -DFW_BUILD_ID explicitly; that value wins
# and the git derivation is skipped. Whether the id describes the copied source
# is the caller's responsibility, and a wrong value still yields a wrongly
# labelled image.
# A trailing '+' means the checkout did not match the commit exactly, whether
# from modified tracked files or new untracked ones, so the hash alone would
# not describe what is actually running. Ignored paths (build trees, a vendored
# SDK) are excluded by `git status --porcelain`, so they never trip it.
#
# An empty -DFW_BUILD_ID counts as unset: passing one through an undefined
# shell variable defines the cache entry as "", which must fall through to the
# derivation rather than silently shipping a bare "FWNX1E-".
if(DEFINED FW_BUILD_ID)
	set(_fw_override "${FW_BUILD_ID}")
else()
	set(_fw_override "")
endif()

# The derivation must run in `cmake -P` script mode at build time, which needs
# it in a file. Materialise it into the build tree at configure time rather
# than tracking a separate .cmake (this keeps the whole mechanism in one place
# and out of the Dockerfile's COPY list) and rather than include()-ing it (in
# -P mode CMAKE_CURRENT_LIST_DIR is the script's own directory, so git would run
# against the wrong tree and always report the fallback). The source tree to
# interrogate is therefore passed in explicitly as GIT_SRC_DIR below.
set(_fw_id_script ${CMAKE_BINARY_DIR}/fw_build_id.cmake)
file(WRITE ${_fw_id_script} [=[
# Generated by CMakeLists.txt; re-run on every build to derive FW_BUILD_ID.
if(DEFINED FW_BUILD_ID_OVERRIDE AND NOT FW_BUILD_ID_OVERRIDE STREQUAL "")
	set(_id "${FW_BUILD_ID_OVERRIDE}")
else()
	set(_id "")
	find_package(Git QUIET)
	if(GIT_FOUND)
		execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse --short=7 HEAD
			WORKING_DIRECTORY "${GIT_SRC_DIR}"
			OUTPUT_VARIABLE _sha OUTPUT_STRIP_TRAILING_WHITESPACE
			ERROR_QUIET RESULT_VARIABLE _rc)
		if(_rc EQUAL 0)
			execute_process(COMMAND ${GIT_EXECUTABLE} status --porcelain
				WORKING_DIRECTORY "${GIT_SRC_DIR}"
				OUTPUT_VARIABLE _dirty OUTPUT_STRIP_TRAILING_WHITESPACE
				ERROR_QUIET)
			if(NOT _dirty STREQUAL "")
				set(_sha "${_sha}+")
			endif()
			set(_id "${_sha}")
		endif()
	endif()
	# git absent or not a repo: fall back to "dev" (soft, not a hard error) to
	# preserve the prior behaviour and keep non-git checkouts building. The
	# Docker path never reaches here -- it always passes an explicit id.
	if(_id STREQUAL "")
		set(_id "dev")
	endif()
endif()
set(_content "#define FW_BUILD_ID \"${_id}\"\n")
# Rewrite only on change, so an unchanged id does not bump the header's mtime
# and force a needless recompile/relink on every build.
set(_old "")
if(EXISTS "${OUT_HEADER}")
	file(READ "${OUT_HEADER}" _old)
endif()
if(NOT _old STREQUAL _content)
	file(WRITE "${OUT_HEADER}" "${_content}")
endif()
]=])

set(_fw_id_header ${CMAKE_BINARY_DIR}/generated/fw_build_id.h)
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/generated)
# Seed the header so the first compile has something to include even if it races
# the custom target under a parallel make; the build-time run overwrites it.
if(NOT EXISTS ${_fw_id_header})
	file(WRITE ${_fw_id_header} "#define FW_BUILD_ID \"dev\"\n")
endif()

add_custom_target(fw_build_id
	COMMAND ${CMAKE_COMMAND}
		-DFW_BUILD_ID_OVERRIDE=${_fw_override}
		-DGIT_SRC_DIR=${CMAKE_CURRENT_LIST_DIR}
		-DOUT_HEADER=${_fw_id_header}
		-P ${_fw_id_script}
	BYPRODUCTS ${_fw_id_header}
	COMMENT "Deriving firmware build id"
	VERBATIM)
message(STATUS "Firmware build id: derived at build time (override='${_fw_override}')")

add_executable(main
	main.c ssd1306.c
	)

# Deliver the build-time id without editing main.c: force-include the generated
# header ahead of the translation unit, so main.c's own `#ifndef FW_BUILD_ID`
# fallback stays dormant and picks up the derived value. OBJECT_DEPENDS makes
# main.c recompile whenever the id changes; add_dependencies orders the
# derivation before the compile. Scoped to main.c -- ssd1306.c never used the id.
add_dependencies(main fw_build_id)
set_source_files_properties(main.c PROPERTIES
	COMPILE_OPTIONS "-include;${_fw_id_header}"
	OBJECT_DEPENDS "${_fw_id_header}")

pico_generate_pio_header(main ${CMAKE_CURRENT_LIST_DIR}/pulse_ts.pio)
pico_enable_stdio_usb(main 1)
pico_enable_stdio_uart(main 1)

# SDK 2.x made PICO_DEFAULT_UART_INSTANCE a function-like macro, so defining it
# to `uart1` broke every translation unit. Select the instance by index instead:
# this spelling works on both SDK 1.5.x and 2.x.
add_compile_definitions(PICO_DEFAULT_UART=1)
add_compile_definitions(PICO_DEFAULT_UART_TX_PIN=4)
add_compile_definitions(PICO_DEFAULT_UART_RX_PIN=5)

pico_add_extra_outputs(main)

target_link_libraries(main
	pico_stdlib hardware_adc hardware_pwm hardware_i2c hardware_pio hardware_dma)
